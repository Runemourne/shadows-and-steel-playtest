def vkey(f): return [int(x) for x in re.findall(r"\d+", f)]
def latest(prefix):
    fs = sorted((f for f in os.listdir(HERE) if re.match(re.escape(prefix) + r"v[\d.]+\.csv$", f)), key=vkey)
    return fs[-1] if fs else None


class Card:
    def __init__(s, r):
        g = lambda k, d=0: int(float(r.get(k) or d))
        s.id, s.name, s.type, s.cls, s.gen, s.sub = r["ID"], r["Name"].strip().upper(), r["Type"], r["Class"], r["Gen"], (r.get("Subtype") or "").title()
        s.tags = set((r.get("Tags") or "").upper().split()); s.cost, s.ready, s.isr, s.timer = g("Cost"), g("Ready", 1), g("ISR"), g("Timer")
        s.conc, s.surv, s.dur, s.atk, s.hf = g("Conc"), g("Surv"), g("Dur"), g("ATK"), g("HF")
        s.range, s.targets, s.req = r.get("Range") or "", r.get("Targets") or "", (r.get("Req") or "").upper().split()
        raw = [x.strip() for x in (r.get("Line 1"), r.get("Line 2"), r.get("Line 3"), r.get("Line 4")) if x and x.strip()]
        s.expend = any(x.rstrip(".") == "EXPEND" or x.rstrip(".").endswith(" EXPEND") for x in raw); s.l = [x for x in raw if x != "EXPEND"]  # EXPEND is printed with the card's costs; a bare EXPEND line (older files) still counts
        s.inv = float(r.get("Inv") or 0); s.copies = g("Copies", 3); s.rep = r.get("Represents") or ""; s.box = r.get("Box") or ""
        s.cap = s.type == "Capability"; s.keys = {}; s.kill = {}
        for tok in (r.get("Keys") or "").split(";"):
            w = tok.split()
            if not w: continue
            if w[0] == "KILL": s.kill[w[1]] = int(w[2])
            else: s.keys[w[0]] = int(w[1]) if len(w) > 1 else 1
        if s.cap:
            for k in s.keys:
                if k not in ABILITY_TEXT: raise KeyError(k)
        s.air = (r.get("Tags") or "").upper().split()[:1] == ["AIR"]; _ut = (r.get("Upgrades From") or "").strip(); s.upfrom = _ut; s.up_cls = None; s.up_tags = set()
        if _ut:   # "[Class or Any] [Generation] ([TAG] or [TAG])": the tag may be a Specialization (strict), a Type, or a Domain (broad); no tag means any card of that Class
            _cn = re.sub(r"\s+(I|II|III)\b.*$", "", _ut.split("(")[0].strip()); s.up_cls = "*" if _cn.lower() == "any" else CLASS_KEY.get(_cn, _cn)
            _m = re.search(r"\(([^)]*)\)", _ut); s.up_tags = set(t.strip().upper() for t in re.split(r"\bor\b|/|,", _m.group(1), flags=re.I) if t.strip()) if _m else set()
        s.gi = {"I": 1, "II": 2, "III": 3}.get(s.gen, 1)
    def lines(s): return list(s.l)
    def cost_text(s):
        if s.cap: return "Deploy %d" % s.cost
        b = ["%d Supply" % s.cost] if s.cost else []
        if s.isr: b.append("%d ISR" % s.isr)
        return " + ".join(b) or "Free"
    def stats(s): return "Conceal %d · Surviv %d · Durab %d%s · Ready %d" % (s.conc, s.surv, s.dur, " · ATK %d %s" % (s.atk, s.range) if s.atk else "", s.ready) if s.cap else ""
    def kind(s): return "Capability" if s.cap else s.sub
    def text(s): return " / ".join(s.l)


def load_cards(path=None):
    f = path or os.path.join(HERE, latest("SS_Merged_Cards_"))
    with open(f, newline="", encoding="utf-8-sig") as fh: rows = list(csv.DictReader(fh))
    return {c.name: c for c in map(Card, rows)}, os.path.basename(f)


def load_decks(cards, path=None):
    f = path or os.path.join(HERE, latest("SS_Merged_Decks_")); out = {}
    with open(f, newline="", encoding="utf-8-sig") as fh:
        for r in csv.DictReader(fh):
            d = out.setdefault(r["Deck"], dict(concept=r["Deck"], box=r["Box"], ie=[], deck=[]))
            d["ie" if r["Part"].startswith("Initial") else "deck"] += [r["Card"].strip().upper()] * int(r["Copies"])
    return out


def deck_cost(cards, names): return round(sum(cards[n].inv for n in names if n in cards), 2)

class Player:
    def __init__(s, i, spec, cards, R, rng, box_i):
        s.i, s.name, s.concept = i, spec.get("name") or spec["concept"], spec["concept"]; s.con = CONCEPTS[s.concept]
        s.ie = [cards[n] for n in spec["ie"]]; s.mf = [cards[n] for n in spec["deck"]]; rng.shuffle(s.ie); rng.shuffle(s.mf)
        s.hand, s.discard, s.ash, s.units, s.track = [], [], [], [], []
        bs = BASES[box_key(spec.get("box") or s.con["box"])][box_i * 2: box_i * 2 + 2]; s.bases = [Base(n, k) for n, k in bs]  # the deck file's Box column decides the box (falls back to the Concept)
        s.ben = s.concept if R["concepts"] else ""; on = 1 if R["concepts"] else 0
        s.hf = R["hf"] + on * s.con.get("hf", 0); s.isr = R["isr0"] + on * s.con.get("isr", 0); s.bases[0].sup += on * s.con.get("sup", 0)
        s.ticks = set(); s.fl = {}; s.lift = 0; s.span_bonus = 0
    def front(s): return [u for u in s.units if u.row == "F"]
    def rear(s): return [u for u in s.units if u.row == "R"]
    def sup(s): return sum(b.sup for b in s.bases)
    def has(s, key): return any(u.k(key) for u in s.units)
    def cls_units(s, *cl): return [u for u in s.units if not u.hidden and u.c.cls in cl]

class Game:
    def __init__(s, a, b, cards, terrain=None, seed=0, first=None, rules=None, log=False):
        s.R = dict(RULES); s.R.update(rules or {}); s.rng = random.Random(seed); s.cards = cards
        s.terrain = terrain or "OPEN PLAIN"; T = TERRAIN[s.terrain]
        for k in ("front_limit", "isr_income"):
            if k in T: s.R[k] = T[k]
        same = box_key(a.get("box") or CONCEPTS[a["concept"]]["box"]) == box_key(b.get("box") or CONCEPTS[b["concept"]]["box"])
        s.P = [Player(0, a, cards, s.R, s.rng, 0), Player(1, b, cards, s.R, s.rng, 1 if same else 0)]
        s.Z = s.R["zones"]; s.pos = [1, s.Z]; s.round = 0; s.first = s.rng.randrange(2) if first is None else first
        s.log = [] if log else None; s.why = [{}, {}]; s.info = 0; s.on_event = None; s.on_decide = None; s.fx = None; s.meta = []; s.endq = []; s.human = None; s.human_pid = None; s.winner = None; s.how = ""; s.phase = "Set up"; s.stats = dict(attacks=0, kills=0, supported=0, effects=0, reactions=0, hidden=0)
        s.P[s.first].isr += 1; s.add_supply(s.P[1 - s.first], 1); s.evl = []
    # ------------------------------------------------------------ helpers
    def ev(s, p, key):   # a note for the probes (who did what, in which round); changes nothing
        if not getattr(s, "_la", False): s.evl.append((s.round, p.i, key))
    def adv(s, p): return p.isr > s.opp(p).isr   # ADVANTAGE: only while you have more ISR than the enemy
    def pin_value(s, x):   # the bot pins the card that matters most: a ready attacker, or the card holding the line when the forces are close

    def attack_options(s, p):
        o = s.opp(p); opts = []
        for u in p.units:
            if u.ex or not u.c.cap or not (u.c.atk or s.hijack_atk(p, u)): continue
            rev = u.hidden
            if rev and (p.sup() < u.c.cost or s.phase != "Operations"): continue

            tg = s.targets(p, u)
            if rev: u.hidden = True; u.fresh = False

                elif t.hidden: isr = s.R["hidden_isr"]; sv = 0; need = 3; name = "a Hidden card in the %s" % ("Frontline" if t.row == "F" else "Rear Area")
                else: isr = s.tconc(p, o, t); sv = max(0, s.tsurv(o, t) - prec); need = t.c.dur - t.dmg; name = t.c.name + (" (Rear Area)" if t.row == "R" else "")
                if c.range != "Close": isr += s.R["ranged_isr"]  # a long shot needs targeting data
                if c.cls == "FIRES": isr = max(0, isr - max([x.k("SHOOTER") for x in p.units if not x.hidden] + [0]))   # KILL WEB: SENSOR-TO-SHOOTER
                sup = s.supporters(p, u, isr, need + sv - a); fs = sum(x.k("FS") for x in sup); cue = sum(x.k("CUE") for x in sup)
                kc = p.ben == "KILL CHAIN" and not p.fl.get("kc") and c.cls == "FIRES" and any(x.c.cls == "INTEL" for x in sup)

                    continue
                hits = max(0, a + fs - max(0, sv - (1 if kc else 0)))
                if kind == "b": last = all(x.dev for x in o.bases if x is not t); v = hits * (1.6 if last else 0.9) + ((60 if last else 7) if hits >= need else 0)

                v -= 0.3 * cost + 0.35 * len(sup) + (0.4 if c.ready > 1 else 0)

                opts.append((v + (0.3 if rev else 0), lab, ("attack", u, kind, t, sup, cost, kc, rev)))
                if s.is_human(p):   # digital game: every legal set of supporters (different Classes, up to Span of Control), and alone
                    for alt in s.support_sets(p, u):
                        if set(alt) == set(sup): continue
                        fs2 = sum(x.k("FS") for x in alt); cue2 = sum(x.k("CUE") for x in alt); cost2 = max(0, isr - cue2)
                        if cost2 > p.isr: continue

                        opts.append((0, "%sttack %s with %s %s(ATK %d, costs %d ISR%s)" % ("Reveal %s for %d Supply and a" % (c.name, c.cost) if rev else "A", name, c.name, ("with support from " + " and ".join(x.c.name for x in alt) + " ") if alt else "alone ", a + fs2, cost2, ", should do %d damage" % h2 if not (kind == "u" and t.hidden) else ""), ("attack", u, kind, t, list(alt), cost2, False, rev)))

    def react(s, p, d, ctx):
        """One reaction by defender d to an action by p. ctx: dict(kind='attack'|'effect', unit=attacker, target=..., tkind, air=bool, ranged=bool, jam=n).
        Returns dict(cancel=bool, cover=n)."""
        res = dict(cancel=False, cover=0); jam = ctx.get("jam", 0)
        if d.isr < jam or ctx.get("noreact"): return res
        opts = [(0.0, "Do not react", None)]; u = ctx.get("unit"); t = ctx.get("target"); atk = ctx["kind"] == "attack"
        threat = ctx.get("threat", 1.0)
        for x in d.units:
            if x.ex or x.hidden or not x.c.cap or x is t: continue
            front = isinstance(t, Unit) and t.row == "F"; rearish = t is not None and not front; noint = ctx.get("noint")
            n = x.k("R_COVER") or x.k("R_SHIELD") or (x.k("R_ESCORT") if front else 0) or (x.k("R_GUARD") if rearish else 0) or (x.k("R_SCREEN") if rearish else 0)
            if x.k("R_SCREEN") and rearish and not noint and atk and u is not None and (u.c.tags & {"DRONE", "ROBOT"}): opts.append((threat + 0.5, "%s: intercept and cancel it" % x.c.name, ("cancel", x)))
            if n and t is not None and ctx.get("hits", 0) > 0: opts.append((min(n, ctx["hits"]) * 0.6 + (2.0 if ctx.get("kill") and ctx["hits"] - n < ctx.get("need", 99) else -0.3), "%s: give the target +%d Survivability" % (x.c.name, n), ("cover", x, n)))
            if not noint and ((x.k("R_SHIELD") and ctx.get("air")) or (x.k("R_ESCORT") and ctx.get("air") and front) or (x.k("R_GUARD") and (ctx.get("air") or ctx.get("rocket")) and rearish)): opts.append((threat + 0.5, "%s: intercept and cancel it" % x.c.name, ("cancel", x)))
            if x.k("R_GLIDE") and not noint and (not atk or ctx.get("ranged")): opts.append((threat + 1.0, "%s: cancel it%s" % (x.c.name, " and deal 2 damage to the attacker" if u else ""), ("glide", x)))
        for c in d.hand:
            if c.sub != "Reaction" or d.sup() < c.cost or d.isr < c.isr + jam or not s.req_ok(d, c): continue
            if "E_HARDKILL" in c.keys and (ctx.get("hits", 0) > 0 or not atk): opts.append((threat + 0.2, "Play HARD KILL (%s): cancel it and ready the protected card" % c.cost_text(), ("card", c)))
            if "E_JAMSWEEP" in c.keys and (not atk or ctx.get("ranged")): opts.append((threat * 0.8 + 0.4, "Play JAMMING SWEEP (%s): cancel it unless the enemy pays 3 ISR; the enemy loses 2 ISR" % c.cost_text(), ("card", c)))
            if "E_WITHDRAW" in c.keys and atk and s.contact() and s.pos[d.i] != (1 if d.i == 0 else s.Z): opts.append((0.3 if ctx.get("hits", 0) > 1 else -1, "Play TACTICAL WITHDRAWAL (%s): fall back 1 Zone, your Frontline gets +1 Survivability" % c.cost_text(), ("card", c)))
            if "E_UNMASK" in c.keys and any(x.hidden for x in p.units): opts.append((1.3, "Play UNMASK (%s): reveal up to 2 enemy Hidden cards and draw 1" % c.cost_text(), ("card", c)))
            if "E_OODA" in c.keys: opts.append((0.9, "Play OODA LOOP (%s): take one attack of your own first" % c.cost_text(), ("card", c)))
        if len(opts) == 1: return res
        ch = s.decide(d, "reaction", "%s Do you react?%s" % (ctx["text"], " (reacting costs %d ISR: jammed)" % jam if jam else ""), opts)
        if not ch: return res
        d.isr -= jam; s.stats["reactions"] += 1
        if ch[0] == "cover": ch[1].ex = True; res["cover"] = ch[2]; s.say(s.who(d) + "'s", ch[1].c.name, "covers the target: +%d Survivability." % ch[2])
        elif ch[0] == "cancel": ch[1].ex = True; res["cancel"] = True; s.say(s.who(d) + "'s", ch[1].c.name, "intercepts. The action is canceled.")
        elif ch[0] == "glide":
            ch[1].ex = True; res["cancel"] = True; s.say(s.who(d) + "'s", ch[1].c.name, "cancels the action" + (" and deals 2 damage to %s." % u.c.name if u is not None else "."))
            if u is not None and not s.R.get("glide_nodmg"): s.hit_unit(p, u, 2)
        else:
            c = ch[1]; s.pay(d, c.cost); d.isr -= c.isr; d.hand.remove(c); d.discard.append(c); s.say(s.who(d), "reacts with", c.name + "."); s.ev(d, next(iter(c.keys)))
            if "E_HARDKILL" in c.keys:
                res["cancel"] = True
                if isinstance(t, Unit): t.ex = False
            elif "E_JAMSWEEP" in c.keys:
                if p.isr >= 3 and s.decide(p, "pay", "JAMMING SWEEP: pay 3 ISR to push your action through?", [(1, "Pay 3 ISR", 1), (0, "Let it be canceled", 0)]): p.isr -= 3; s.say(s.who(p), "pays 3 ISR to push through the jamming.")
                else: res["cancel"] = True
                s.gain_isr(p, -2)
            elif "E_WITHDRAW" in c.keys: s.pos[d.i] += -1 if d.i == 0 else 1; d.fl["wd"] = 1; s.say(s.who(d), "falls back 1 Zone, breaking Contact."); res["wd"] = True
            elif "E_UNMASK" in c.keys:
                for x in [x for x in p.units if x.hidden][:2]:
                    if not s.reveal(p, x, forced=True): s.gain_isr(d, 3)
                s.draw(d, 1)
            elif "E_OODA" in c.keys:
                o2 = s.attack_options(d)
                if o2:
                    pick = s.decide(d, "action", "OODA LOOP: choose your attack.", o2)
                    if pick: s.do_attack(d, pick, noreact=True)
        return res

    def req_ok(s, p, c):
        r = c.req
        if not r: return True
        if r[0] == "CONTACT": return s.contact()
        if r[0] == "CULMINATE": return s.contact() and sum(1 for u in s.opp(p).front() if u.ex) >= 2
        if r[0] == "UNIT": return bool(p.cls_units(*r[1:]))
        if r[0] == "UNITS2": return len(p.cls_units(*r[1:])) >= 2
        if r[0] == "PULSE": return any(not b.dev and not b.ex and (set(r[1:-1]) & set(b.t()["access"])) for b in p.bases)
        return True

    def collapse_n(s, p): return max(0, min(s.R["collapse_cap"], p.isr - s.opp(p).isr))
    def collapse_value(s, p):   # the bot fires COLLAPSE OF WILL when it wins the game, at a gap of collapse_min or more, or late with a gap of 2 or more
        n = s.collapse_n(p); o = s.opp(p)

    def look_ahead(s, p, m, base=0.0):
        """Play the rest of the round a few times for one choice of move and score the result. Every choice is tried against the same dice."""
        tot = 0.0; n = s.R.get("la_samples", 2)
        for k in range(n):
            g = copy.deepcopy(s); g.human = None; g.log = None; g.on_event = None; g.on_decide = None; g._la = True; g.rng = random.Random(base + k); q = g.P[p.i]

                if q.i == g.first: g.arrange(g.opp(q)); g.maneuver(g.opp(q))
                g.operations(); g.sustainment(); g.assessment()
            except Over: pass
            tot += g.score(q)
        return tot / float(n)

                    s.say("H-Hour:", s.who(p) + "'s", c.name, "completes its mission."); u = Unit(c, "R"); u.ex = True; u.dmg = t.get("dmg", 0); p.units.append(u)
                    op = t.get("op", "INF_PSYOP"); o = s.opp(p); s.ev(p, "H:" + op)
                    if op == "INF_HIT":
                        tg = [x for x in o.units if x.c.cap and (s.R.get("op_anydomain", 1) or "LAND" in x.c.tags)]
                        if tg: x = max(tg, key=lambda x: s.unit_value(x) + x.dmg); s.say(c.name, "strikes", x.c.name if not x.hidden else "a Hidden card", "for %d damage." % t["n"]); s.hit_unit(o, x, t["n"])
                    elif op == "INF_INTEL": s.gain_isr(p, t["n"]); s.draw(p, 1); s.say(c.name, "reports in: +%d ISR and a card." % t["n"])
                    elif op == "INF_REVEAL":
                        s.gain_isr(p, 2)
                        for x in [x for x in o.units if x.hidden][:t["n"]]: s.reveal(o, x, forced=True)
                    elif t.get("sab"): s.raid(p, t["sab"], c.name)
                    else: s.lose_hf(o, t.get("n", 2), c.name)
                    for _ in range(t.get("dis", 0)): s.cancel_track(o, c.name)
                    ot = [x for x in s.opp(p).track if x["t"] < 3]
                    if ot and t.get("op", "INF_PSYOP") == "INF_PSYOP": ot[0]["t"] += 1
                    continue
                s.say("H-Hour:", s.who(p) + "'s", c.name, ("resolves." if t["kind"] == "op" else "ends.")); s.ev(p, "H:" + next(iter(c.keys)))
                (p.ash if c.expend else p.discard).append(c); s.resolve(p, c)
        passes = 0; i = 0; guard = 0
        while passes < 2 and guard < 80:
            guard += 1; p = order[i % 2]; i += 1
            passes = 0 if s.take_action(p) else passes + 1
    def sustainment(s):
        s.phase = "Sustainment"; R = s.R
        for p in s.P:
            o = s.opp(p); s.gain_isr(p, R["isr_income"] + (1 if sum(1 for u in o.units if not u.hidden) - sum(1 for u in p.units if not u.hidden) >= 2 else 0))
            for u in p.units:
                if u.k("CONVOY"): s.add_supply(p, u.k("CONVOY"))
            for b in p.bases: b.sup = min(b.stock(), b.sup + b.restock()); b.ex = False
        for p in s.P:
            free = 1 if s.own_half(p) else 0; m2 = p.ben == "SPEARHEAD"; reactor = p.has("REACTOR"); rom = sum(u.k("ROM") for u in p.units)
            ex = sorted([u for u in p.units if u.ex and not u.hidden and not u.pin], key=lambda u: -(s.unit_value(u) + u.c.atk)); paid = []   # a PINNED card does not ready at this Recovery
            for u in [u for u in p.units if not u.hidden and (u.k("ADV_DRAW") or u.k("ADV_SUPPLY"))]:
                if s.adv(p): s.say(s.who(p) + "'s", u.c.name + ": ADVANTAGE."); s.draw(p, u.k("ADV_DRAW")); s.add_supply(p, u.k("ADV_SUPPLY")); s.ev(p, "ADVANTAGE")
            for u in ex:
                cost = 0 if reactor and not u.k("REACTOR") else u.c.ready
                if cost and free and u.row == "F": cost = 0; free = 0
                elif cost and m2 and u.c.cls == "M2": cost = 0; m2 = False
                elif cost and rom > 0 and (u.c.cls == "M2" or u.c.air):
                    cost -= 1; rom -= 1; farp = next((x.c.name for x in p.units if x.k("ROM")), "FARP")
                    if cost == 0: s.say(s.who(p) + "'s", farp, "readies", u.c.name, "for 1 less Supply (free).")
                if cost == 0: u.ex = False; continue
                if s.is_human(p): paid.append([u, cost, cost < u.c.ready and not (reactor and not u.k("REACTOR"))]); continue   # digital game: asked below, all at once
                if p.sup() < cost:
                    if u.c.air: p.units.remove(u); p.discard.append(u.c); s.say(s.who(p) + "'s", u.c.name, "returns to base (discarded, no Homefront loss).")
                    continue
                keep = 1 if (s.unit_value(u) + u.c.atk >= 3.5 or p.sup() - cost >= 4) else -1
                disc = cost < u.c.ready and not (reactor and not u.k("REACTOR"))
                if s.decide(p, "ready", "Sustainment: ready %s for %d Supply%s? You have %d Supply.%s" % (u.c.name, cost, " (%s: 1 less)" % next((x.c.name for x in p.units if x.k("ROM")), "FARP") if disc else "", p.sup(), " If you do not, this aircraft returns to base (discarded)." if u.c.air else ""), [(keep, "Pay %d and ready %s" % (cost, u.c.name), u), (0, "Leave it", 0)]): s.pay(p, cost); u.ex = False
                elif u.c.air: p.units.remove(u); p.discard.append(u.c); s.say(s.who(p) + "'s", u.c.name, "returns to base (discarded, no Homefront loss).")
            if paid:   # digital game: the person picks which cards to ready, in any order; aircraft not readied return to base
                fname = next((x.c.name for x in p.units if x.k("ROM")), "FARP"); air = [x[0].c.name for x in paid if x[0].c.air]
                while True:
                    can = [x for x in paid if p.sup() >= x[1]]
                    if not can: break
                    ch = s.decide(p, "ready_select", "Sustainment: pick the cards to ready. Each costs Supply (shown on the button); you have %d Supply.%s" % (p.sup(), (" Aircraft you do not ready return to base (discarded): %s." % ", ".join(air)) if air else ""),
                                  [(1, "Pay %d and ready %s%s" % (x[1], x[0].c.name, " (%s: 1 less)" % fname if x[2] else ""), x[0]) for x in can] + [(0, "Done readying", 0)])
                    if not ch: break
                    x = next(y for y in paid if y[0] is ch); s.pay(p, x[1]); ch.ex = False; paid.remove(x)
                    if ch.c.name in air: air.remove(ch.c.name)
                for u, c0, d0 in paid:
                    if u.c.air: p.units.remove(u); p.discard.append(u.c); s.say(s.who(p) + "'s", u.c.name, "returns to base (discarded, no Homefront loss).")
            for u in sorted([u for u in p.units if u.dmg and not u.hidden], key=lambda u: -(u.dmg + s.unit_value(u))):
                while u.dmg and p.sup() >= R["recover_cost"] + (0 if s.is_human(p) else 4) and s.decide(p, "ready", "Recover 1 damage on %s for %d Supply? You have %d Supply." % (u.c.name, R["recover_cost"], p.sup()), [(1 if u.c.dur - u.dmg <= 2 else -1, "Pay and remove 1 damage", 1), (0, "No", 0)]): s.pay(p, R["recover_cost"]); u.dmg -= 1
        for p in s.P:
            for u in p.units: u.spot = False; u.over = False; u.fresh = False; u.rl = False; u.rt = False
    def assessment(s):
        s.phase = "Assessment"; a, b = s.P
        s.stats["contact_rounds"] = s.stats.get("contact_rounds", 0) + (1 if s.contact() else 0)
        if s.contact():
            sa, sb = s.strength(a), s.strength(b)
            if sa == sb: sa += 0.1 * len({u.c.cls for u in a.front() if not u.hidden}); sb += 0.1 * len({u.c.cls for u in b.front() if not u.hidden})
            s.say("Frontline contest: %s %d, %s %d." % (s.who(a), s.strength(a), s.who(b), s.strength(b)))
            for h, e, vh, ve in ((a, b, sa, sb), (b, a, sb, sa)):  # count last stands: a contest fought with a marker in its own end Zone
                if s.pos[h.i] == (1 if h.i == 0 else s.Z):
                    bonus = s.R["home_bonus"] or sum(1 for x in h.bases if not x.dev); s.stats["home_contests"] = s.stats.get("home_contests", 0) + 1
                    if vh >= ve: s.stats["home_held"] = s.stats.get("home_held", 0) + 1
                    if vh >= ve and vh - bonus < ve: s.stats["home_saved"] = s.stats.get("home_saved", 0) + 1
            if sa != sb:
                w, l = (a, b) if sa > sb else (b, a); n = s.R["contest_retreat"]
                rt = max([u.k("ROUT") for u in w.units if u.row == "F" and not u.hidden] + [1 if w.fl.get("rout") else 0])
                if rt and (w.fl.get("noobs") or not s.obstacle(l)): n += rt
                if any(u.k("IMMOVABLE") and u.row == "F" and not u.hidden for u in l.units) and s.decide(l, "reaction", "BREAKWATER: NEVER RETREAT. Hold your ground instead of falling back %d Zone(s)?" % n, [(1, "Never retreat: hold", 1), (0, "Fall back", 0)]): n = 0; s.say(s.who(l), "holds the line: NEVER RETREAT.")
                home = 1 if l.i == 0 else s.Z; room = abs(s.pos[l.i] - home)
                if room == 0 and s.R["supremacy"]: s.say(s.who(l), "is pushed off the home edge."); s.end(w, "Supremacy: the enemy was driven off its home edge", True)
                n = min(n, room); s.pos[l.i] += -n if l.i == 0 else n
                s.say(s.who(w), "wins the contest.", s.who(l), "retreats %d Zone%s." % (n, "" if n == 1 else "s")); s.lose_hf(l, s.R["contest_hf"], "lost the Frontline contest")
                if n: s.pursue(w, l)
                ex = max([u.k("EXPLOIT") for u in w.front() if not u.hidden] + [0])
                if ex and s.winner is None: s.raid(w, ex, "EXPLOITATION")
        if s.R["centre"]:
            c = (s.Z + 1) / 2.0; ha, hb = s.pos[0] >= c, s.pos[1] <= c
            if ha != hb:
                w = a if ha else b; s.say(s.who(w), "holds the centre of the route.")
                if s.cond(s.opp(w), "E_NARR"): s.say("STRATEGIC NARRATIVE: no Homefront is lost for the centre.")
                else: s.lose_hf(s.opp(w), s.R["centre"], "the enemy holds the centre")
        for p in s.P:
            o = s.opp(p)
            if p.ben == "SPECTRUM" and p.isr >= o.isr + s.R.get("spec_gap", 3): s.lose_hf(o, 1, "DECISION ADVANTAGE")
            if s.cond(p, "E_PSYOP"): s.lose_hf(o, 1, "PSYOP CAMPAIGN")
            if s.cond(p, "E_INFOADV") and p.isr >= o.isr + s.R.get("infoadv_gap", 1): s.lose_hf(o, 1, "INFORMATION ADVANTAGE")
        wc = s.R["war_cost"] + (1 if s.R.get("escalate", 0) and s.round >= s.R["escalate"] else 0)   # ESCALATION: the cost of war rises late in the war
        for p in s.P: p.hf -= wc
        s.say("The cost of war: each side loses %d Homefront (%s %d, %s %d)." % (wc, a.name, a.hf, b.name, b.hf))
        for p in s.P: s.tick(p)
        wins = {q[0].i for q in s.endq if q[1].startswith("Devastation")}
        for p in s.P:
            if all(x.dev for x in p.bases): wins.add(1 - p.i)
            if p.hf <= 0: wins.add(1 - p.i)
        if len(wins) == 1 and min(a.hf, b.hf) > 0: w = s.P[wins.pop()]; s.end(w, "Devastation: both Bases destroyed", True)
        if wins or s.round >= s.R["last"]:
            if a.hf == b.hf: s.end(None, "Both sides exhausted: a draw", True)
            s.end(a if a.hf > b.hf else b, "Homefront collapse" if min(a.hf, b.hf) <= 0 else ("Higher Homefront" if wins else "Higher Homefront after round %d" % s.round), True)
        s.endq = []
        for p in s.P:
            p.fl = {}
            for u in p.units: u.pin = False   # PIN ends with the round; the card is still exhausted, and readies at the next Recovery as usual
            if s.R["forced_discard"] and p.hand: p.discard += p.hand; p.hand = []
        if s.R["forced_discard"]: s.say("Both players discard their hands.")
    def play(s):
        try:
            for p in s.P: s.draw(p, 0)
            while True:
                s.round += 1; s.say("--- Round %d. %s is the Active Player. ---" % (s.round, s.P[s.first].name)); order = [s.P[s.first], s.P[1 - s.first]]
                for p in order: s.command(p)
                for p in order: s.phase = "Command"; s.arrange(p); s.maneuver(p)
                s.operations(); s.sustainment(); s.assessment(); s.first = 1 - s.first
        except Over: pass
        s.say("Game over:", ("a draw" if s.winner == -1 else s.P[s.winner].name + " wins") + ".", s.how)
        return s.winner
