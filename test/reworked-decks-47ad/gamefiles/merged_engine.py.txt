def whole_number_seed(fraction):
    """The whole number that a 64-bit Python seeds random.Random with for this fraction (zero or more): its hash.
    Seeding with it gives the same dice on every Python; the browser's 32-bit Python hashes a fraction differently (F10)."""
    top, bottom = fraction.as_integer_ratio()
    prime = 2 ** 61 - 1
    return top % prime * pow(bottom, prime - 2, prime) % prime

class Card:
    def __init__(s, r):
        g = lambda k, d=0: int(float(r.get(k) or d))
        s.id, s.name, s.type, s.cls, s.gen, s.sub = r["ID"], r["Name"].strip().upper(), r["Type"], r["Class"], r["Gen"], (r.get("Subtype") or "").title()
        if s.sub == "Operation": s.sub = "Plan"   # v0.17.4 (AN.4): the subtype is called Plan; a card file that still says Operation is read as Plan
        s.tags = set((r.get("Tags") or "").upper().split()); s.cost, s.ready, s.isr, s.timer = g("Cost"), g("Ready", 1), g("ISR"), g("Timer")
        s.conc, s.surv, s.dur, s.atk, s.hf = g("Conc"), g("Surv"), g("Dur"), g("ATK"), g("HF")
        s.range, s.targets, s.req = r.get("Range") or "", r.get("Targets") or "", (r.get("Req") or "").upper().split()
        raw = [x.strip() for x in (r.get("Line 1"), r.get("Line 2"), r.get("Line 3"), r.get("Line 4")) if x and x.strip()]
        s.expend = any(x.rstrip(".") == "EXPEND" or x.rstrip(".").endswith(" EXPEND") for x in raw); s.l = [x for x in raw if x != "EXPEND"]  # EXPEND is printed with the card's costs; a bare EXPEND line (older files) still counts
        s.inv = float(r.get("Inv") or 0); s.copies = g("Copies", 3); s.rep = r.get("Represents") or ""; s.box = r.get("Box") or ""
        s.cap = s.type == "Capability"; s.keys = {}; s.kill = {}
        for tok in (r.get("Keys") or "").split(";"):
            w = tok.split()
            if not w: continue
            if w[0] == "KILL": s.kill[w[1]] = int(w[2])
            else: s.keys[w[0]] = int(w[1]) if len(w) > 1 else 1
        if s.cap:
            for k in s.keys:
                if k not in ABILITY_TEXT and k not in TRIAL_TEXT: raise KeyError(k)   # Sprint 3 trial keys (TRIAL_TEXT)
        s.air = (r.get("Tags") or "").upper().split()[:1] == ["AIR"]; _ut = (r.get("Upgrades From") or "").strip(); s.upfrom = _ut; s.up_cls = None; s.up_tags = set()
        if _ut:   # "[Class or Any] [Generation] ([TAG] or [TAG])": the tag may be a Specialization (strict), a Type, or a Domain (broad); no tag means any card of that Class
            _cn = re.sub(r"\s+(I|II|III)\b.*$", "", _ut.split("(")[0].strip()); s.up_cls = "*" if _cn.lower() == "any" else CLASS_KEY.get(_cn, _cn)
            _m = re.search(r"\(([^)]*)\)", _ut); s.up_tags = set(t.strip().upper() for t in re.split(r"\bor\b|/|,", _m.group(1), flags=re.I) if t.strip()) if _m else set()
        s.gi = {"I": 1, "II": 2, "III": 3}.get(s.gen, 1)
    def lines(s): return list(s.l)
    def action_isr(s, ability):
        """The ISR printed in the costs of this card's ACTION line for the named ability ("ACTION [1 ISR][EXHAUST] · SPOOF:"), or 0."""
        for line in s.l:
            printed = re.match(r"ACTION\s*((?:\[[^\]]*\])*)\s*·\s*" + re.escape(ability) + r":", line)
            if printed:
                cost = re.search(r"\[(\d+) ISR\]", printed.group(1))
                return int(cost.group(1)) if cost else 0
        return 0
    def action_name(s, effect, otherwise):
        """The name this card's ACTION line prints for the ability whose words hold `effect` ("ACTION [EXHAUST] · NAME 2: effect"), or otherwise."""
        for line in s.l:
            printed = re.match(r"ACTION\s*(?:\[[^\]]*\])*\s*·\s*([^:]*?)(?:\s+\d+)?:(.*)", line)
            if printed and effect in printed.group(2): return printed.group(1)
        return otherwise
    def cost_text(s):
        if s.cap: return "Deploy %d" % s.cost
        b = ["%d Supply" % s.cost] if s.cost else []
        if s.isr: b.append("%d ISR" % s.isr)
        return " + ".join(b) or "Free"
    def stats(s): return "Conceal %d · Surviv %d · Durab %d%s · Ready %d" % (s.conc, s.surv, s.dur, " · ATK %d %s" % (s.atk, s.range) if s.atk else "", s.ready) if s.cap else ""
    def kind(s): return "Capability" if s.cap else s.sub
    def text(s): return " / ".join(s.l)

def load_decks(cards, path=None):
    f = path or os.path.join(HERE, latest("SS_Merged_Decks_")); out = {}
    with open(f, newline="", encoding="utf-8-sig") as fh:
        for r in csv.DictReader(fh):

SCENARIO_KEYS = dict(game=("first", "last", "sides"), side=("homefront", "isr", "bases", "marker", "frontline", "rear"),
                     base=("supply", "damage", "devastated"))
def refuse_unknown(what, given, allowed):
    unknown = sorted(set(given) - set(allowed))
    if unknown: raise ValueError("The scenario's %s has keys the engine does not know: %s." % (what, ", ".join(unknown)))
def whole_number(what, value, lowest, highest):
    """A scenario's number, refused in plain words unless it is a whole number from lowest to highest."""
    if isinstance(value, bool) or not isinstance(value, int) or not lowest <= value <= highest:
        raise ValueError("The scenario's %s must be a whole number from %d to %d, not %r." % (what, lowest, highest, value))
    return value
def checked_scenario(scenario):
    """A mission's starting situation (the Sprint 7 contract, section 3). A key the engine does not know is refused, never ignored."""
    scenario = dict(scenario or {}); refuse_unknown("game", scenario, SCENARIO_KEYS["game"])
    if scenario.get("first") not in (None, 0, 1): raise ValueError("The scenario's first Active Player must be seat 0 or 1.")
    if scenario.get("first") is None: scenario.pop("first", None)
    if len(scenario.get("sides", [])) > 2: raise ValueError("The scenario has more than two sides.")
    for side in scenario.get("sides", []):
        refuse_unknown("side", side, SCENARIO_KEYS["side"])
        if len(side.get("bases", [])) > 2: raise ValueError("A side of the scenario has more than two Bases.")
        for base in side.get("bases", []):
            refuse_unknown("Base", base, SCENARIO_KEYS["base"])
            if base.get("devastated") not in (None, True, False): raise ValueError("A Base's devastated must be true or false.")
    return scenario


class Game:
    def __init__(s, a, b, cards, terrain=None, seed=0, first=None, rules=None, log=False, scenario=None):
        s.R = dict(RULES); s.R.update(rules or {}); s.rng = random.Random(seed); s.cards = cards
        s.rngs = s.new_rngs(seed) if s.R.get("rng_split", 0) else None   # trial rng_split: separate dice (None = the one stream, as released)
        s.terrain = terrain or "OPEN PLAIN"; T = TERRAIN[s.terrain]
        for k in ("front_limit", "isr_income"):
            if k in T: s.R[k] = T[k]
        same = box_key(a.get("box") or CONCEPTS[a["concept"]]["box"]) == box_key(b.get("box") or CONCEPTS[b["concept"]]["box"])
        s.P = [Player(0, a, cards, s.R, s.rs("deck", 0), 0), Player(1, b, cards, s.R, s.rs("deck", 1), 1 if same else 0)]   # rng_split: each player's own shuffle dice
        s.Z = s.R["zones"]; s.pos = [1, s.Z]; s.round = 0; s.first = s.rng.randrange(2) if first is None else first
        s.log = [] if log else None; s.why = [{}, {}]; s.info = 0; s.on_event = None; s.on_decide = None; s.fx = None; s.meta = []; s.endq = []; s.human = None; s.human_pid = None; s.winner = None; s.how = ""; s.phase = "Set up"; s.stats = dict(attacks=0, kills=0, supported=0, effects=0, reactions=0, hidden=0)
        s.result = None; s.scenario = checked_scenario(scenario); s.first = s.scenario.get("first", s.first)
        s.give_the_start_bonus(); s.evl = []; s.set_up_the_scenario()
    # ------------------------------------------------------------ Sprint 7: the start bonus and a mission's starting situation
    def give_the_start_bonus(s):
        """Rule start_bonus: 0 (as released) the first player gains 1 ISR and the second 1 Supply; 1 (Rulebook 3.6.5, section 4, and
        the lead's original rule, drafts entries H, I and K) the second player draws 1 more card and the first gets nothing; 2 none."""
        if s.R.get("start_bonus", 0) == 0: s.P[s.first].isr += 1; s.add_supply(s.P[1 - s.first], 1)
    def start_card(s, player):   # start_bonus 1: the second player's one more card, drawn with its first hand in round 1
        second = player is s.P[1 - s.first] and s.round == 1
        return 1 if second and s.R.get("start_bonus", 0) == 1 else 0
    def set_up_the_scenario(s):
        """A mission's starting situation (the Sprint 7 contract, section 3), applied after the start bonus."""
        if "last" in s.scenario: s.R["last"] = whole_number("last round", s.scenario["last"], 1, 99)
        for player, side in zip(s.P, s.scenario.get("sides", [])): s.set_up_side(player, side)
    def set_up_side(s, player, side):
        if "homefront" in side: player.hf = whole_number("Homefront", side["homefront"], 1, s.R["hf"])
        if "isr" in side: player.isr = whole_number("ISR", side["isr"], 0, s.R["isr_cap"])
        states = list(zip(player.bases, side.get("bases", [])))
        for base, state in states:   # Bases that start devastated first, their Supply moving as in play; then the numbers as stated
            if state.get("devastated"): s.devastate_at_start(player, base)
        for base, state in states: s.set_up_base(base, state)
        if "marker" in side: s.place_marker(player, whole_number("Force Marker's Zone", side["marker"], 1, s.Z))
        for row, key in (("F", "frontline"), ("R", "rear")):
            for name in side.get(key, []): s.put_on_the_table(player, name, row)
    def devastate_at_start(s, player, base):   # as in play: Supply over what a devastated Base holds moves to the other Base
        other = next(other for other in player.bases if other is not base); base.dev = True; base.dmg = base.t()["dur"]
        extra = max(0, base.sup - base.stock()); base.sup = min(base.sup, base.stock()); other.sup = min(other.stock(), other.sup + extra)
    def set_up_base(s, base, state):
        if "supply" in state and whole_number("Supply", state["supply"], 0, 999) > base.stock():
            words = "a devastated Base holds" if base.dev else "it holds"
            raise ValueError("%s cannot start with %d Supply: %s at most %d." % (base.name, state["supply"], words, base.stock()))
        if "supply" in state: base.sup = state["supply"]
        if "damage" in state and base.dev: raise ValueError("%s starts devastated: give it no damage as well." % base.name)
        if "damage" in state: base.dmg = whole_number("damage on %s" % base.name, state["damage"], 0, base.t()["dur"] - 1)
    def place_marker(s, player, zone):
        others = s.pos[1 - player.i]; ahead = zone <= others if player.i == 0 else zone >= others
        if not 1 <= zone <= s.Z or not ahead: raise ValueError("A Force Marker cannot start in Zone %d: it would leave the route or pass the enemy marker." % zone)
        s.pos[player.i] = zone
    def put_on_the_table(s, player, name, row):
        """A card that starts on the table: face up, Ready and undamaged; a copy is taken from the side's decks if one is there."""
        card = s.cards.get(name)
        if card is None or not card.cap: raise ValueError("%s is not a Capability in the card file." % name)
        if row == "F" and len(player.front()) >= s.R["front_limit"]: raise ValueError("%s: the Frontline is full." % name)
        for pile in (player.ie, player.mf):
            copy_in_deck = next((other for other in pile if other.name == name), None)
            if copy_in_deck is not None: pile.remove(copy_in_deck); break
        player.units.append(Unit(card, row))
    # ------------------------------------------------------------ helpers
    def new_rngs(s, x): return dict(deck=[random.Random("%r deck %d" % (x, i)) for i in (0, 1)], bot=random.Random("%r bot" % x), la=random.Random("%r la" % x))   # rng_split: one stream per purpose
    def rs(s, what, i=None):   # the dice for one purpose: the one stream (rng_split 0, as released) or that purpose's own
        r = getattr(s, "rngs", None); return s.rng if not r else (r[what][i] if i is not None else r[what])
    def ev(s, p, key):   # a note for the probes (who did what, in which round); changes nothing
        if not getattr(s, "_la", False): s.evl.append((s.round, p.i, key))
    def adv(s, p): return p.isr > s.opp(p).isr   # ADVANTAGE: only while you have more ISR than the enemy
    def pin_value(s, x):   # the bot pins the card that matters most: a ready attacker, or the card holding the line when the forces are close
        return (0.3 * x.c.atk + 0.8 if not x.ex else 0) + ((x.k("HOLD") or 1) * (0.9 if s.dist() == 0 else (0.4 if s.dist() == 1 else 0.1)) if s.R["pin_str"] else 0) + 0.05 * s.unit_value(x)
    def card_worth(s, c): return (c.cost + 0.5 * c.atk + c.hf + (3 if c.keys.get("FLAGSHIP") else 0)) if c.cap else (1.0 + c.cost + c.isr + (1 if c.req[:1] == ["PULSE"] else 0))
    def compromise(s, p, n, src):
        """COMPROMISE n: p looks at the top n cards of the enemy Main Force, discards any of them and puts the rest back (in any order)."""
        o = s.opp(p); top = o.mf[-n:] if n > 0 else []
        if not top: return 0
        s.info += 1; out = []
        if s.is_human(p):   # a person picks, one at a time, and may stop
            pool = list(top)
            while pool:
                ch = s.decide(p, "compromise", "%s: COMPROMISE %d. Discard which of the enemy's top Main Force cards? (%d discarded)" % (src, n, len(out)), [(1, "Discard %s (%s)" % (c.name, c.kind()), c) for c in dict.fromkeys(pool)] + [(0, "Done", 0)])

    def support_sets(s, p, u):

    def supporters(s, p, u, isr_cost, hits_needed):

    def take_action(s, p):
        opts = s.attack_options(p) + s.ability_options(p) + s.effect_options(p)
        if not s.is_human(p): opts = [o for o in opts if o[0] > 0.15]
        opts.append((0.1, "Pass", None))
        ch = s.decide(p, "action", "Operations, round %d: choose an action. You have %d Supply and %d ISR." % (s.round, p.sup(), p.isr), opts)
        if ch is None: s.say(s.who(p), "passes."); return False
        s.one_act(p.i)   # Sprint 4 sust_onereact: this action, for the one-Reaction rule
        if ch[0] == "attack": s.do_attack(p, ch)
        elif ch[0] == "effect": s.play_effect(p, ch[1])
        elif ch[0] == "ab":
            if s.do_ability(p, ch[1], ch[2], len(ch) > 3 and ch[3]): s.one_end(); return s.take_action(p) or True
        s.one_end()
        return True
    # ------------------------------------------------------------ the round's printed steps, told beside the log
    def step_begins(s, number, player=None):
        """Tell the listener (on_event) that printed sub-step `number` of ROUND_STEPS begins, for `player` where each player plays it.
        An event beside the log, never a line in it; silent in the bot's look-ahead copies and when nobody listens."""
        s.step_ends()
        if not s.on_event or getattr(s, "_la", False): return
        step = ROUND_STEPS[number - 1]; who = "" if player is None else ", " + player.name
        event = dict(step, kind="step", round=s.round, player=None if player is None else player.i, player_name=None if player is None else player.name)
        s.open_step = dict(event, before=s.what_both_sides_see())
        s.on_event("Round %d. %s, step %d: %s%s." % (s.round, step["phase"], step["in_phase"], step["name"], who), event)
    def step_ends(s):
        """The step told last is over: tell its account, if it has one (STEP_ACCOUNTS), from what both sides can see."""
        step = getattr(s, "open_step", None); s.open_step = None
        if not step or not s.on_event or getattr(s, "_la", False) or step["number"] not in s.STEP_ACCOUNTS: return
        account = getattr(s, s.STEP_ACCOUNTS[step["number"]])
        s.tell_account(step, account(step, step["before"], s.what_both_sides_see()))
    def tell_account(s, step, text):
        """What the game did in one step, as a plain sentence beside the log. It may name only what both sides can see."""
        if not s.on_event or getattr(s, "_la", False): return
        event = dict((key, value) for key, value in step.items() if key != "before"); event.update(kind="step_account", text=text)
        s.on_event(text, event)
    def what_both_sides_see(s):
        """The open table, for the step accounts: each side's Homefront, ISR, Supply, rows and damage; a face-down card by its place only."""
        def label(unit): return "a Hidden card" if unit.hidden else unit.c.name
        return [dict(hf=side.hf, isr=side.isr, supply=side.sup(), rows=dict((unit.uid, (unit.row, label(unit))) for unit in side.units),
                     damage=dict((unit.uid, (unit.dmg, label(unit))) for unit in side.units if not unit.hidden)) for side in s.P]
    STEP_ACCOUNTS = {4: "arrange_account", 8: "replenish_account", 10: "recover_account", 13: "concepts_and_conditions_account"}   # step 1 tells its own (discard_and_draw)
    def names_or_none(s, names): return ", ".join(names) if names else "none"
    def arrange_account(s, step, before, after):
        side = s.P[step["player"]]; was, now = before[side.i]["rows"], after[side.i]["rows"]
        moved_up = [now[uid][1] for uid in now if now[uid][0] == "F" and was.get(uid, ("F",))[0] != "F"]
        moved_back = [now[uid][1] for uid in now if now[uid][0] == "R" and was.get(uid, ("R",))[0] != "R"]
        rows = "Frontline: %s. Rear Area: %s." % tuple(s.names_or_none([now[uid][1] for uid in now if now[uid][0] == row]) for row in ("F", "R"))
        if not moved_up and not moved_back: return "%s: rows unchanged. %s" % (side.name, rows)

    def look_ahead(s, p, m, base=0.0):

                g.operations(); g.sustainment(); g.assessment()
            except Over: pass
            tot += g.score(q)

                s.say(s.who(w), "holds the centre of the route.")

                else: s.lose_hf(s.opp(w), s.R["centre"], "the enemy holds the centre")
    def the_center_as_printed(s, holder):
        """Step 12's line with rule as_printed: who loses, and whose Force Marker alone has reached the center (American spelling)."""
        other = s.opp(holder); lost = s.R["centre"]
        if s.cond(other, "E_NARR"):
            narrative = next(entry["c"].name for entry in other.track if "E_NARR" in entry["c"].keys)
            return s.say("%s loses no Homefront for the center: its %s is in effect." % (s.who(other), narrative))
        told = "%s loses %d Homefront: only %s's Force Marker has reached the center." % (s.who(other), lost, s.who(holder))
        s.lose_hf(other, lost, "the enemy holds the centre", told=told)
    def concepts_and_conditions(s):
        """Assessment, step 3: the Concepts and Conditions that work at Assessment."""
        s.step_begins(13)
        for p in s.P:
            o = s.opp(p)
            if p.ben == "SPECTRUM" and p.isr >= o.isr + s.R.get("spec_gap", 3): s.lose_hf(o, 1 + (1 if s.R.get("spec_zero", 0) and o.isr <= 0 else 0), "DECISION ADVANTAGE")
            if s.cond(p, "E_PSYOP"): s.lose_hf(o, 1, "PSYOP CAMPAIGN")
            if s.cond(p, "E_INFOADV") and p.isr >= o.isr + s.R.get("infoadv_gap", 1): s.lose_hf(o, 1, "INFORMATION ADVANTAGE")
    def play(s):
        try:
            for p in s.P: s.draw(p, 0)
            while True:
                s.round += 1; s.say("--- Round %d. %s is the Active Player. ---" % (s.round, s.P[s.first].name)); order = [s.P[s.first], s.P[1 - s.first]]
                if s.R.get("command_together", 0): s.draw_together(order)
                for p in order: s.command(p)
                for p in order: s.phase = "Command"; s.arrange(p); s.maneuver(p)
                s.operations(); s.sustainment(); s.assessment(); s.first = 1 - s.first
        except Over: pass
        s.step_ends()
        s.say("Game over:", ("a draw" if s.winner == -1 else s.P[s.winner].name + " wins") + ".", s.how)
        return s.winner
    # ------------------------------------------------------------ what a player may see

                             lines=c.lines() if c.cap else ["Decoy: " + c.name], id=c.id, ready=c.ready, cost=c.cost, hf=c.hf, tags=" ".join(sorted(c.tags)), sub=c.sub)
        def side(p, own):

        return v
